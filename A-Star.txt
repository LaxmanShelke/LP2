import heapq

goal = (
    (1, 2, 3),
    (4, 5, 6),
    (7, 8, 0)
)

def print_state(state):
    for row in state:
        print(row)

def misplaced_tiles(state):

    count = 0

    for i in range(3):
        for j in range(3):

            if state[i][j] != 0 and state[i][j] != goal[i][j]:
                count += 1

    return count

def get_neighbors(state):

    neighbors = []

    for i in range(3):
        for j in range(3):

            if state[i][j] == 0:
                x, y = i, j

    directions = [(-1,0), (1,0), (0,-1), (0,1)]

    for dx, dy in directions:

        nx, ny = x + dx, y + dy

        if 0 <= nx < 3 and 0 <= ny < 3:

            temp = [list(row) for row in state]

            temp[x][y], temp[nx][ny] = temp[nx][ny], temp[x][y]

            neighbors.append(tuple(tuple(row) for row in temp))

    return neighbors

def is_solvable(state):

    arr = []

    for row in state:
        for num in row:
            if num != 0:
                arr.append(num)

    inversions = 0

    for i in range(len(arr)):
        for j in range(i + 1, len(arr)):

            if arr[i] > arr[j]:
                inversions += 1

    return inversions % 2 == 0

def a_star(start):

    start = tuple(tuple(row) for row in start)

    pq = []

    h = misplaced_tiles(start)

    heapq.heappush(pq, (h, 0, start, [(start, 0, h)]))

    visited = set()

    while pq:

        f, g, current, path = heapq.heappop(pq)

        if current in visited:
            continue

        visited.add(current)

        if current == goal:

            print("\nSolution Found\n")

            for step, (state, g_val, h_val) in enumerate(path):

                print("Step", step)

                print_state(state)

                print("g(n) =", g_val)

                print("h(n) =", h_val)

                print("f(n) =", g_val + h_val)

                print("-------------------")

            return

        for neighbor in get_neighbors(current):

            if neighbor not in visited:

                new_g = g + 1

                new_h = misplaced_tiles(neighbor)

                new_f = new_g + new_h

                heapq.heappush(
                    pq,
                    (new_f, new_g, neighbor,
                    path + [(neighbor, new_g, new_h)])
                )

    print("No Solution")

Artificial Intelligence A* (A-Star) Algorithm is one of the most popular searching algorithms used for path finding and game searching problems.

It finds the shortest and optimal path between:
Start node
Goal node

A* combines:

Actual path cost
Estimated future cost

to efficiently search the best path.

A* Evaluation Function

f(n)=g(n)+h(n)

Where:
g(n) = Actual cost from start node to current node
h(n) = Heuristic estimated cost from current node to goal
f(n) = Total estimated cost
-----------------------
Working of A* Algorithm
Start from source node
Add source node into open list
Select node with lowest f(n)
Expand neighbouring nodes
Calculate cost values
Repeat until goal node is reached

A* Algorithm Steps:

Step 1:Initialize open list and closed list.

Step 2:Add start node to open list.

Step 3:Select node having minimum f(n).

Step 4:If goal node found, stop.

Step 5:Generate neighbouring nodes.

Step 6:Calculate g(n), h(n), and f(n).

Step 7:Repeat until goal reached.
